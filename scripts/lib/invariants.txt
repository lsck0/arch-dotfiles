# the declared end state, one fact per line, asserted by dot test run's verify inside the VM (every row) and by
# dot status on the live machine (the live rows). one definition, so both check the same facts.
# <key> <want> <where> <probe>: where is live (both) or vm (the test only: it needs the install password, or root
# would be needed to see a real answer). the probe runs in bash as the admin, read-only, and prints the value; n/a is a
# fact this platform does not have (no efi under wsl, no cups), skipped live and a mismatch in the VM.
# @PASSWORD@ and @DISK@ are filled in by dot test.
stage_armed       no   live  test -e /var/lib/dotfiles-stage/next && echo yes || echo no
sudo_passwordless no   vm    sudo -n true 2>/dev/null && echo yes || echo no
keyfile_present   no   vm    test -e /etc/cryptsetup-keys.d/root.key && echo yes || echo no
luks_keyslots     1    vm    echo @PASSWORD@ | sudo -S cryptsetup luksDump @DISK@2 2>/dev/null | grep -cE '^  [0-9]+: luks2'
secure_boot       1    live  [ -d /sys/firmware/efi ] || { echo n/a; exit 0; }; od -An -t u1 /sys/firmware/efi/efivars/SecureBoot-8be4df61-93ca-11d2-aa0d-00e098032b8c | awk '{print $NF}'
setup_mode        0    live  [ -d /sys/firmware/efi ] || { echo n/a; exit 0; }; od -An -t u1 /sys/firmware/efi/efivars/SetupMode-8be4df61-93ca-11d2-aa0d-00e098032b8c | awk '{print $NF}'
kernel_modules    yes  live  [ "$(systemd-detect-virt 2>/dev/null)" != wsl ] || { echo n/a; exit 0; }; test -d "/usr/lib/modules/$(uname -r)" && echo yes || echo no
pdf_queue         yes  live  command -v lpstat >/dev/null || { echo n/a; exit 0; }; lpstat -p PDF >/dev/null 2>&1 && echo yes || echo no
